async function createJob(){ const message = document.getElementById("customerMessage"); message.innerText = "Submitting request..."; try { /* GET LOGGED-IN CUSTOMER */ const { data: { user }, error: authError } = await supabaseClient.auth.getUser(); if(authError){ throw new Error( "Authentication error: " + authError.message ); } if(!user){ throw new Error( "You are not logged in. Please login again." ); } console.log("Logged-in customer ID:", user.id); /* GET FORM VALUES */ const service = document.getElementById("serviceType") .value; const description = document.getElementById("jobDescription") .value.trim(); const location = document.getElementById("jobLocation") .value.trim(); const date = document.getElementById("preferredDate") .value; const time = document.getElementById("preferredTime") .value.trim(); const budget = Number( document.getElementById("customerBudget") .value ); /* VALIDATION */ if( !service || !description || !location || !budget ){ throw new Error( "Please complete the service, description, location and budget." ); } /* INSERT REQUEST */ const { data, error } = await supabaseClient .from("service_requests") .insert({ customer_id: user.id, service_type: service, description: description, location: location, phone: null, preferred_date: date || null, preferred_time: time || null, budget: budget, customer_budget: budget, status: "pending" }) .select() .single(); /* DATABASE ERROR */ if(error){ console.error( "SUPABASE REQUEST ERROR:", error ); throw new Error( "Supabase error: " + error.message ); } /* SUCCESS */ console.log( "REQUEST CREATED:", data ); message.innerText = "✅ Request submitted successfully!"; await loadMyRequests(); await loadProviders(service); } catch(error){ console.error( "REQUEST SUBMISSION FAILED:", error ); message.innerText = "❌ " + error.message; alert( "Request could not be submitted.\n\n" + error.message ); } }
