async function loadUserDashboard(userId) { try { // Get the user's profile. // limit(1) prevents duplicate profile rows // from breaking the login. const { data: profiles, error } = await supabaseClient .from("profiles") .select("id, full_name, email, role") .eq("id", userId) .limit(1); if (error) { console.error("Profile error:", error); throw new Error( "Your account logged in, but your profile could not be loaded." ); } let profile = profiles && profiles.length > 0 ? profiles[0] : null; /* * If no profile exists, create one from * the information stored in the user's account. */ if (!profile) { const { data: userData, error: userError } = await supabaseClient.auth.getUser(); if (userError || !userData.user) { throw new Error("User account could not be found."); } const user = userData.user; const fullName = user.user_metadata?.full_name || user.email?.split("@")[0] || "Customer"; const role = user.user_metadata?.role || "customer"; const { data: newProfile, error: createError } = await supabaseClient .from("profiles") .insert({ id: user.id, full_name: fullName, email: user.email, role: role }) .select() .limit(1); if (createError) { console.error("Create profile error:", createError); // Even if profile creation has an issue, // continue using the user's metadata. profile = { id: user.id, full_name: fullName, email: user.email, role: role }; } else { profile = newProfile[0]; } } /* * Hide everything first. */ document.getElementById("home").style.display = "none"; document .getElementById("customerDashboard") .classList.remove("active"); document .getElementById("providerDashboard") .classList.remove("active"); /* * PROVIDER */ if (profile.role === "provider") { document .getElementById("providerDashboard") .classList.add("active"); return; } /* * CUSTOMER */ document.getElementById("customerName").innerText = profile.full_name || "Customer"; document .getElementById("customerDashboard") .classList.add("active"); } catch (error) { console.error(error); alert( "Login succeeded, but your dashboard could not be loaded.\n\n" + error.message ); } }
